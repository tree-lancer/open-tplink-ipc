cmake_minimum_required(VERSION 3.16)
project(tplink_ipc_mcp VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(LibXml2 REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(tplink-ipc-mcp src/main.cpp)
target_compile_definitions(tplink-ipc-mcp PRIVATE TPLINK_IPC_VERSION="${PROJECT_VERSION}")
target_link_libraries(tplink-ipc-mcp PRIVATE CURL::libcurl OpenSSL::Crypto LibXml2::LibXml2 nlohmann_json::nlohmann_json)
target_compile_options(tplink-ipc-mcp PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS tplink-ipc-mcp RUNTIME DESTINATION bin)
